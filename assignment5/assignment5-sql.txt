SELECT orders.order_id, ROUND(SUM(line_items.quantity * products.price)::numeric, 2) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.order_id ORDER BY orders.order_id LIMIT 5;
SELECT c.customer_name, ROUND(AVG(t.total_price)::numeric, 2) AS average_order_price FROM customers c JOIN (SELECT orders.customer_id, orders.order_id, SUM(line_items.quantity * products.price) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.customer_id, orders.order_id) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name;
